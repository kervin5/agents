# Python code to calculate the first 10,000 terms of the series and multiply by 4
total_sum = 0

for n in range(10000):
    term = 1 / (2 * n + 1)  # Calculate the term using the formula
    if n % 2 == 0:
        total_sum += term  # Add the term for even indices
    else:
        total_sum -= term  # Subtract the term for odd indices

final_result = total_sum * 4  # Multiply the total by 4
print(final_result)
